        (function() {
          function initSvcCards() {
            var cards = document.querySelectorAll('.svc-card');
            if (!cards.length) return;
            function toggle(card) {
              var wasActive = card.classList.contains('active');
              cards.forEach(function(c) { c.classList.remove('active', 'dimmed'); c.setAttribute('aria-expanded', 'false'); });
              if (!wasActive) {
                card.classList.add('active');
                card.setAttribute('aria-expanded', 'true');
                cards.forEach(function(c) { if (c !== card) c.classList.add('dimmed'); });
              }
            }
            cards.forEach(function(card) {
              card.addEventListener('click', function(e) {
                if (e.target.closest('a') || e.target.closest('.svc-card-more')) return;
                toggle(card);
              });
              card.addEventListener('keydown', function(e) {
                if ((e.key === 'Enter' || e.key === ' ') && e.target === card) { e.preventDefault(); toggle(card); }
              });
            });
            document.querySelectorAll('.svc-chip').forEach(function(chip) {
              chip.addEventListener('click', function() {
                var t = document.querySelector(chip.getAttribute('href'));
                if (t && !t.classList.contains('active')) setTimeout(function() { t.click(); }, 150);
              });
            });
          }
          if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', initSvcCards); }
          else { initSvcCards(); }
        })();
        </script>
